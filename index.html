import streamlit as st import pandas as pd # 1. 페이지 기본 설정 및 제목 st.set_page_config(page_title="거래처 데이터 조회", layout="wide") st.title("📊 거래처 정보 필터링 대시보드") st.markdown("드롭다운 박스를 선택하여 원하는 거래처 정보를 빠르게 필터링하세요.") # 2. CSV 데이터 불러오기 # 파일이 app.py와 같은 폴더에 있다고 가정합니다. @st.cache_data def load_data(): # 첫 번째 열이 비어있거나 인덱스인 경우를 고려해 index_col=0 지정 가능 df = pd.read_csv("0608_중복점.csv", index_col=0) # 주소나 다른 열의 공백 제거 및 결측치 채우기 df['거래처명'] = df['거래처명'].fillna('이름 없음') df['P코드'] = df['P코드'].fillna('코드 없음') df['시도명'] = df['시도명'].fillna('미지정') return df try: df = load_data() # 3. 사이드바 또는 메인화면에 드롭다운(선택 박스) 배치 st.subheader("🔍 필터 조건 선택") # 레이아웃을 3개의 칸으로 나누어 드롭다운 배치 col1, col2, col3 = st.columns(3) with col1: # '시도명' 고유값 추출 (전체 보기 옵션 추가) sido_list = ["전체"] + sorted(df['시도명'].unique().tolist()) selected_sido = st.selectbox("📍 지역(시도) 선택", sido_list) with col2: # 선택한 시도에 따라 거래처명 필터링하여 드롭다운에 표시 if selected_sido != "전체": filtered_names = df[df['시도명'] == selected_sido]['거래처명'].unique() else: filtered_names = df['거래처명'].unique() name_list = ["전체"] + sorted(filtered_names.tolist()) selected_name = st.selectbox("🏪 거래처명 선택", name_list) with col3: # P코드 검색 및 선택 드롭다운 if selected_sido != "전체": filtered_pcodes = df[df['시도명'] == selected_sido]['P코드'].unique() else: filtered_pcodes = df['P코드'].unique() pcode_list = ["전체"] + sorted(filtered_pcodes.tolist()) selected_pcode = st.selectbox("🔑 P코드 선택", pcode_list) # 4. 선택된 드롭다운 조건에 따라 데이터 최종 필터링 filtered_df = df.copy() if selected_sido != "전체": filtered_df = filtered_df[filtered_df['시도명'] == selected_sido] if selected_name != "전체": filtered_df = filtered_df[filtered_df['거래처명'] == selected_name] if selected_pcode != "전체": filtered_df = filtered_df[filtered_df['P코드'] == selected_pcode] # 5. 결과 화면 출력 st.write("---") st.subheader(f"📋 조회 결과 (총 {len(filtered_df)}건)") # 보여주고 싶은 핵심 컬럼만 순서대로 배치 display_cols = ['P코드', '거래처명', '시도명', '시군구명', '읍면동명', '주소', '대리점명', '실적'] # 존재하는 컬럼만 안전하게 출력 valid_cols = [col for col in display_cols if col in filtered_df.columns] if not filtered_df.empty: st.dataframe(filtered_df[valid_cols], use_container_width=True) else: st.warning("선택한 조건에 일치하는 데이터가 없습니다.") except FileNotFoundError: st.error("❌ '0608_중복점.csv' 파일을 찾을 수 없습니다. 코드 파일과 같은 폴더에 넣어주세요.")

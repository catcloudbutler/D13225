function displayStoreMarkersOnMap(data) { storeCustomOverlays.forEach(overlay => overlay.setMap(null)); storeCustomOverlays = []; if (currentOpenedOverlay) { currentOpenedOverlay.setMap(null); currentOpenedOverlay = null; } if (!map || data.length === 0) return; const validStores = data.filter(item => item.lat > 0 && item.lng > 0); const coordGroups = new Map(); validStores.forEach(store => { const coordKey = `${store.lat.toFixed(5)}_${store.lng.toFixed(5)}`; if (!coordGroups.has(coordKey)) { coordGroups.set(coordKey, []); } coordGroups.get(coordKey).push(store); }); // 🚀 [수정] 기존 .slice(0, 100)을 .slice(0, 2000)으로 변경하여 주변 거래처가 모두 나오도록 확장 const uniqueCoords = Array.from(coordGroups.keys()).slice(0, 2000); uniqueCoords.forEach(coordKey => { const storesAtCoord = coordGroups.get(coordKey); const representativeStore = storesAtCoord[0]; const markerPosition = new kakao.maps.LatLng(representativeStore.lat, representativeStore.lng); const isGJ = storesAtCoord.some(s => s.agency.includes('광장대리점')); const isPSM = storesAtCoord.some(s => s.agency.toUpperCase().includes('PS&M') || s.agency.includes('피에스앤마케팅')); let pinClass = "pin-normal"; if (isGJ) { pinClass = "pin-gj"; } else if (isPSM && storesAtCoord.some(s => s.type === 'PS_전속')) { pinClass = "pin-psm-exclusive"; } else if (isPSM && storesAtCoord.some(s => s.type === '중복점')) { pinClass = "pin-psm-duplicate"; } const contentEl = document.createElement('div'); contentEl.className = `custom-pin-marker ${pinClass}`; const customOverlay = new kakao.maps.CustomOverlay({ position: markerPosition, content: contentEl, yAnchor: 1.0, xAnchor: 0.5 }); customOverlay.setMap(map); storeCustomOverlays.push(customOverlay); contentEl.addEventListener('click', function() { if (currentOpenedOverlay) { currentOpenedOverlay.setMap(null); } const repAddress = `${representativeStore.sido} ${representativeStore.sigungu} ${representativeStore.address}`.trim(); const storeNameMap = new Map(); storesAtCoord.forEach(store => { const storeUniqueId = `${store.name}_${store.pcode}`; if (!storeNameMap.has(storeUniqueId)) { storeNameMap.set(storeUniqueId, { name: store.name, pcode: store.pcode, type: store.type, agencies: [] }); } const currentObj = storeNameMap.get(storeUniqueId); if (!currentObj.agencies.some(a => a.name === store.agency)) { currentObj.agencies.push({ name: store.agency, score: store.score }); } }); const popupEl = document.createElement('div'); popupEl.className = 'popup-wrap'; let bodyHtml = `
📍 ${repAddress}
`; storeNameMap.forEach((info) => { const totalScore = info.agencies.reduce((sum, item) => sum + item.score, 0); let agencyRows = info.agencies.map(a => `
  • ${a.name} (${a.score.toLocaleString()}건)
    • `).join(''); bodyHtml += `
    🏢 ${info.name} ${info.pcode}
    중복여부: ${info.type}
    📋 거래대리점명(실적):
      ${agencyRows}
    실적 합계: ${totalScore.toLocaleString()}건
    `; }); bodyHtml += `
    `; popupEl.innerHTML = bodyHtml; popupEl.querySelector('.popup-close-btn').onclick = function() { popupOverlay.setMap(null); if (currentOpenedOverlay === popupOverlay) { currentOpenedOverlay = null; } }; const preventMapEvents = function(e) { if (e.stopPropagation) { e.stopPropagation(); } else { e.cancelBubble = true; } }; popupEl.addEventListener('wheel', preventMapEvents, { passive: true }); popupEl.addEventListener('mousewheel', preventMapEvents, { passive: true }); popupEl.addEventListener('DOMMouseScroll', preventMapEvents, { passive: true }); popupEl.addEventListener('touchstart', preventMapEvents, { passive: true }); popupEl.addEventListener('touchmove', preventMapEvents, { passive: true }); popupEl.addEventListener('touchend', preventMapEvents, { passive: true }); const popupOverlay = new kakao.maps.CustomOverlay({ position: markerPosition, content: popupEl, yAnchor: 1.0, xAnchor: 0.5 }); popupOverlay.setMap(map); currentOpenedOverlay = popupOverlay; }); }); }
